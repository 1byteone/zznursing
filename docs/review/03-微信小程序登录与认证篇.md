```java
// FamilyMemberServiceImpl.java - 登录业务实现
@Service
public class FamilyMemberServiceImpl 
    extends ServiceImpl<FamilyMemberMapper, FamilyMember>
    implements IFamilyMemberService {

    @Autowired
    private WechatService wechatService;

    @Autowired
    private TokenService tokenService;

    // 默认昵称前缀列表
    static List<String> DEFAULT_NICKNAME_PREFIX = ListUtil.of(
        "生活更美好", "大桔大利", "日富一日", "好柿开花",
        "柿柿如意", "一椰暴富", "大柚所为", "杨梅吐气", "天生荔枝"
    );

    @Override
    public LoginVo login(UserLoginRequestDto requestDto) {
        // Step 1: 调用微信API获取openId
        String openId = wechatService.getOpenid(requestDto.getCode());
        log.info("获取到openId: {}", openId);

        // Step 2: 根据openId查询用户
        FamilyMember familyMember = getOne(
            Wrappers.<FamilyMember>lambdaQuery()
                .eq(FamilyMember::getOpenId, openId)
        );

        // Step 3: 如果用户不存在，创建新用户（登录即注册）
        if (ObjectUtils.isEmpty(familyMember)) {
            familyMember = FamilyMember.builder()
                .openId(openId)
                .build();
            log.info("新用户，openId: {}", openId);
        }

        // Step 4: 获取手机号
        String phone = wechatService.getPhone(requestDto.getPhoneCode());
        log.info("获取到手机号: {}", phone);

        // Step 5: 保存或更新用户信息
        saveOrUpdateFamilyMember(familyMember, phone);

        // Step 6: 生成JWT Token
        Map<String, Object> claims = new HashMap<>();
        claims.put("userId", familyMember.getId());
        claims.put("nickName", familyMember.getName());
        String token = tokenService.createToken(claims);

        // Step 7: 构建返回对象
        return LoginVo.builder()
            .token(token)
            .nickName(familyMember.getName())
            .build();
    }

    /**
     * 保存或更新家属信息
     */
    private void saveOrUpdateFamilyMember(FamilyMember member, String phone) {
        // 更新手机号
        if (ObjectUtil.notEqual(phone, member.getPhone())) {
            member.setPhone(phone);
        }

        // 已存在用户，更新
        if (ObjectUtil.isNotEmpty(member.getId())) {
            updateById(member);
            return;
        }

        // 新用户，生成随机昵称
        String nickName = DEFAULT_NICKNAME_PREFIX.get(
            (int) (Math.random() * DEFAULT_NICKNAME_PREFIX.size())
        ) + phone.substring(phone.length() - 4);

        member.setName(nickName);
        save(member);
    }
}

// WechatServiceImpl.java - 微信API服务
@Service
@Slf4j
public class WechatServiceImpl implements WechatService {

    private static final String JSCODE2SESSION_URL =
        "https://api.weixin.qq.com/sns/jscode2session?grant_type=authorization_code";

    private static final String TOKEN_URL =
        "https://api.weixin.qq.com/cgi-bin/token?grant_type=client_credential";

    private static final String PHONE_URL =
        "https://api.weixin.qq.com/wxa/business/getuserphonenumber?access_token=";

    @Value("${wechat.appId}")
    private String appid;

    @Value("${wechat.appSecret}")
    private String secret;

    @Override
    public String getOpenid(String code) {
        Map<String, Object> paramMap = new HashMap<>();
        paramMap.put("appid", appid);
        paramMap.put("secret", secret);
        paramMap.put("js_code", code);

        String result = HttpUtil.get(JSCODE2SESSION_URL, paramMap);
        JSONObject jsonObject = JSONUtil.parseObj(result);

        // 检查错误
        Integer errcode = jsonObject.getInt("errcode");
        if (ObjectUtil.isNotEmpty(errcode) && errcode != 0) {
            log.error("获取openid失败, errcode: {}, errmsg: {}",
                errcode, jsonObject.getStr("errmsg"));
            throw new BaseException("微信登录失败: " + jsonObject.getStr("errmsg"));
        }

        return jsonObject.getStr("openid");
    }

    @Override
    public String getPhone(String phoneCode) {
        // 1. 获取access_token
        String accessToken = getAccessToken();

        // 2. 调用获取手机号API
        String url = PHONE_URL + accessToken;
        Map<String, Object> paramMap = new HashMap<>();
        paramMap.put("code", phoneCode);

        String result = HttpUtil.post(url, JSONUtil.toJsonStr(paramMap));
        JSONObject jsonObject = JSONUtil.parseObj(result);

        // 3. 检查错误
        Integer errcode = jsonObject.getInt("errcode");
        if (errcode != null && errcode != 0) {
            log.error("获取手机号失败, errcode: {}, errmsg: {}",
                errcode, jsonObject.getStr("errmsg"));
            throw new BaseException("获取手机号失败: " + jsonObject.getStr("errmsg"));
        }

        // 4. 返回手机号
        return jsonObject.getJSONObject("phone_info").getStr("phoneNumber");
    }

    private String getAccessToken() {
        Map<String, Object> paramMap = new HashMap<>();
        paramMap.put("appid", appid);
        paramMap.put("secret", secret);

        String result = HttpUtil.get(TOKEN_URL, paramMap);
        JSONObject jsonObject = JSONUtil.parseObj(result);

        if (ObjectUtil.isNotEmpty(jsonObject.getInt("errcode"))) {
            throw new BaseException("获取access_token失败: " + jsonObject.getStr("errmsg"));
        }

        return jsonObject.getStr("access_token");
    }
}
```

**详细解析**：

1. **微信登录完整链路**：
   ```
   小程序 wx.login()  →  获取临时code（5分钟有效）
   后端 jscode2session  →  获取openid + session_key
   后端 getPhoneNumber  →  获取手机号（需access_token）
   后端 openid查用户  →  存在则更新，不存在则创建
   后端 生成JWT  →  返回token给前端
   ```

2. **登录即注册的设计优势**：
   - 无需额外注册页面
   - 用户首次登录自动创建账号
   - 降低用户使用门槛

3. **昵称生成策略**：
   - 随机前缀（情感化词组） + 手机号后4位
   - 保证唯一性同时增加趣味性
   - 用户后续可自行修改

**面试思维链**：

```
面试官问：小程序登录是怎么实现的？

思维链：
1. 先说整体链路：code → openid → 手机号 → JWT

3. 最后说后续认证：拦截器解析Token + ThreadLocal

回答要点：
"小程序登录是一条链：code → openid → 手机号 → JWT

关键设计点：
1. 登录即注册：首次登录自动创建账号，无需额外注册
2. openid作为唯一标识：微信保证同一小程序的openid唯一
3. 手机号完善信息：通过微信API获取绑定手机号

追问应对：
Q: 如果微信API调用失败怎么办？
A: 1. 记录日志并提示用户稍后重试
   2. access_token有缓存，减少调用频率
   3. 关键操作可降级为手动输入手机号
```

```
  ┌──────────┐    强引用    ┌──────────────────┐

                            │  │ 弱引用-->Key  │ │
                            │  │ 强引用-->Value│ │

                                   │
                    弱引用          │        强引用
              ┌────────────────────┘    ┌───────────┘
              ▼                         ▼

     │  ThreadLocal    │      │  Value          │
     │  对象 (Key)     │      │  userId=100     │
     └─────────────────┘      └─────────────────┘

  GC后 (ThreadLocal对象被回收, 但Value无法回收):
  ┌──────────┐    强引用    ┌──────────────────┐

                            └──────┬─────────────┘

                                   │  强引用仍在!
                                   ▼
                          ┌─────────────────┐
                          │  Value          │
                          │  userId=100     │  ★ 内存泄漏! Value无法被GC回收 ★
```

```java
// UserThreadLocal.java - 用户上下文工具类
@Slf4j
public class UserThreadLocal {

    // static final：所有实例共享，不可变
    private static final ThreadLocal<Long> LOCAL = new ThreadLocal<>();

    // 私有构造器，防止实例化
    private UserThreadLocal() {
    }

    /**
     * 将用户ID放入当前线程的ThreadLocal
     */
    public static void set(Long userId) {
        LOCAL.set(userId);
        log.debug("ThreadLocal已设置用户ID: {}", userId);
    }

    /**
     * 从当前线程的ThreadLocal获取用户ID
     */
    public static Long get() {
        return LOCAL.get();
    }

    /**
     * 获取当前用户ID（便捷方法）
     * @return 用户ID，如果未登录返回null
     */
    public static Long getUserId() {
        return LOCAL.get();
    }

    /**
     * 清除ThreadLocal中的数据
     * 必须调用，防止内存泄漏！
     */
    public static void remove() {
        LOCAL.remove();
        log.debug("ThreadLocal已清除");
    }
}

// MemberInterceptor.java - 会员拦截器
@Component
@Slf4j
public class MemberInterceptor implements HandlerInterceptor {

    @Autowired
    private TokenService tokenService;

    /**
     * 前置处理：解析Token，存储用户ID到ThreadLocal
     *
     * 执行时机：Controller方法执行前
     */
    @Override
    public boolean preHandle(HttpServletRequest request,
                            HttpServletResponse response,
                            Object handler) throws Exception {

        // 1. 如果不是Controller层的请求，直接放行
        if (!(handler instanceof HandlerMethod)) {
            return true;
        }

        // 2. 获取请求头中的Token
        String token = request.getHeader("authorization");
        if (StringUtils.isEmpty(token)) {
            response.setStatus(HttpServletResponse.SC_UNAUTHORIZED);
            log.warn("认证失败: Token为空, URI: {}", request.getRequestURI());
            throw new BaseException("认证失败，请重新登录");
        }

        // 3. 解析Token
        Claims claims = tokenService.parseToken(token);
        if (ObjectUtil.isEmpty(claims)) {
            response.setStatus(HttpServletResponse.SC_UNAUTHORIZED);
            log.warn("认证失败: Token解析失败, token: {}", token);
            throw new BaseException("认证失败，Token无效");
        }

        // 4. 获取Token中的用户ID
        Long userId = MapUtil.get(claims, "userId", Long.class);
        if (ObjectUtil.isEmpty(userId)) {
            response.setStatus(HttpServletResponse.SC_UNAUTHORIZED);
            log.warn("认证失败: Token中无用户ID");
            throw new BaseException("认证失败，用户信息无效");
        }

        // 5. 将用户ID放入ThreadLocal ★核心步骤★
        UserThreadLocal.set(userId);
        log.debug("认证成功, userId: {}, URI: {}", userId, request.getRequestURI());

        return true;
    }

    /**
     * 后置处理：清理ThreadLocal
     *
     * 执行时机：请求完成后（无论成功或异常）
     * 重要性：★★★★★ 必须执行，否则导致内存泄漏和数据污染！
     */
    @Override
    public void afterCompletion(HttpServletRequest request,
                               HttpServletResponse response,
                               Object handler,
                               Exception ex) throws Exception {
        // 必须清理！防止：
        // 1. 内存泄漏（线程池复用导致ThreadLocal数据不释放）
        // 2. 数据污染（下一个请求拿到上一个请求的用户ID）
        UserThreadLocal.remove();
    }
}
```

**详细解析**：

1. **ThreadLocal核心原理**：
   ```
   Thread-1  →  ThreadLocalMap  →  {UserThreadLocal(弱引用): 12345}
   Thread-2  →  ThreadLocalMap  →  {UserThreadLocal(弱引用): 67890}
   Thread-3  →  ThreadLocalMap  →  {UserThreadLocal(弱引用): null}
   ```

2. **内存泄漏的完整链路**：
   ```
   ThreadLocal对象（强引用）
       ↓ 被ThreadLocalMap中的Entry弱引用
   Entry extends WeakReference<ThreadLocal>
       ↓ Entry对Value是强引用
   Value（用户ID等数据）
       ↓ 线程不销毁，ThreadLocalMap不回收
   内存泄漏发生！

   解决：手动调用remove()，清除Entry
   ```

3. **为什么在afterCompletion中清理**：
   - `afterCompletion`无论请求成功或异常都会执行
   - 比`postHandle`更可靠（异常时postHandle可能不执行）
   - 保证100%清理

4. **为什么用static final**：
   - `static`：所有实例共享同一个ThreadLocal对象（作为Key）
   - `final`：防止被意外重新赋值
   - 节省内存：不需要每个实例创建一个ThreadLocal

```java
// TokenService.java - JWT Token服务
@Component
@Slf4j
public class TokenService {

    @Value("${token.secret}")
    private String secret;

    @Value("${token.expireTime:720}")  // 默认720分钟（12小时）
    private int expireTime;

    // Token黑名单前缀（Redis）
    private static final String TOKEN_BLACKLIST_PREFIX = "token:blacklist:";

    @Autowired(required = false)
    private RedisTemplate<Object, Object> redisTemplate;

    /**
     * 创建Token（小程序端）
     */
    public String createToken(Map<String, Object> claims) {
        return createToken(claims, expireTime);
    }

    /**
     * 创建Token（带过期时间）
     */
    public String createToken(Map<String, Object> claims, int expireMinutes) {
        Date now = new Date();
        Date expiration = new Date(now.getTime() + expireMinutes * 60 * 1000L);

        String token = Jwts.builder()
                .setClaims(claims)                              // 设置自定义数据
                .setIssuedAt(now)                               // 签发时间
                .setExpiration(expiration)                      // 过期时间
                .setIssuer("zzyl-nursing")                      // 签发者
                .setSubject(String.valueOf(claims.get("userId"))) // 主题
                .signWith(SignatureAlgorithm.HS512, secret)     // 签名算法
                .compact();

        log.debug("Token已创建, userId: {}, 过期时间: {}", claims.get("userId"), expiration);
        return token;
    }

    /**
     * 解析Token
     */
    public Claims parseToken(String token) {
        try {
            return Jwts.parser()
                    .setSigningKey(secret)
                    .parseClaimsJws(token)
                    .getBody();
        } catch (ExpiredJwtException e) {
            log.warn("Token已过期: {}", e.getMessage());
            throw new BaseException("登录已过期，请重新登录");
        } catch (SignatureException e) {
            log.warn("Token签名无效: {}", e.getMessage());
            throw new BaseException("Token无效");
        } catch (MalformedJwtException e) {
            log.warn("Token格式错误: {}", e.getMessage());
            throw new BaseException("Token格式错误");
        } catch (Exception e) {
            log.error("Token解析失败: {}", e.getMessage());
            throw new BaseException("认证失败");
        }
    }

    /**
     * 刷新Token
     * 在旧Token过期前，用旧Token换取新Token
     */
    public String refreshToken(String oldToken) {
        // 1. 先验证旧Token是否在黑名单中（已登出的Token不能刷新）
        if (isTokenBlacklisted(oldToken)) {
            throw new BaseException("Token已失效，请重新登录");
        }

        // 2. 解析旧Token
        Claims claims = parseToken(oldToken);

        // 3. 生成新Token（延长有效期）
        Map<String, Object> newClaims = new HashMap<>();
        newClaims.put("userId", claims.get("userId"));
        newClaims.put("nickName", claims.get("nickName"));

        String newToken = createToken(newClaims);

        // 4. 将旧Token加入黑名单
        addToBlacklist(oldToken, claims.getExpiration());

        log.info("Token已刷新, userId: {}", claims.get("userId"));
        return newToken;
    }

    /**
     * 登出（将Token加入黑名单）
     */
    public void logout(String token) {
        Claims claims = parseToken(token);
        addToBlacklist(token, claims.getExpiration());
        log.info("用户已登出, userId: {}", claims.get("userId"));
    }

    /**
     * 检查Token是否在黑名单中
     */
    private boolean isTokenBlacklisted(String token) {
        if (redisTemplate == null) {
            return false;
        }
        String key = TOKEN_BLACKLIST_PREFIX + DigestUtil.md5Hex(token);
        return Boolean.TRUE.equals(redisTemplate.hasKey(key));
    }

    /**
     * 将Token加入黑名单
     */
    private void addToBlacklist(String token, Date expiration) {
        if (redisTemplate == null) {
            return;
        }
        String key = TOKEN_BLACKLIST_PREFIX + DigestUtil.md5Hex(token);
        // 黑名单有效期与Token剩余有效期一致
        long remainingTime = expiration.getTime() - System.currentTimeMillis();
        if (remainingTime > 0) {
            redisTemplate.opsForValue().set(key, "1", remainingTime, TimeUnit.MILLISECONDS);
        }
    }
}
```

```
2. 再说安全措施：HTTPS + 过期时间 + 黑名单 + 不存敏感信息
3. 最后说局限性：Payload可读不可改

回答要点：
"JWT的安全性需要从多个层面保障：
1. 传输层：必须使用HTTPS，防止中间人攻击
2. Token本身：Payload不要存敏感信息（只是Base64编码）
3. 时效性：设置合理的过期时间（当前12小时）
4. 退出机制：登出时将Token加入Redis黑名单
5. 签名验证：HS512签名算法，防止Payload被篡改

JWT的核心优势是无状态，服务端不需要存储Session，
但这也意味着无法主动让Token失效（除非用黑名单）。"

追问应对：
Q: JWT和Session有什么区别？
A: JWT无状态，服务端不存储；Session有状态，服务端需要存储。JWT适合分布式，Session适合单机

Q: Token过期了怎么处理？
A: 1. 前端拦截401错误，跳转登录页
   2. 可设计Token刷新机制，避免频繁登录
   3. 敏感操作需要重新登录
```

```java
// 增强版TokenService
@Component
@Slf4j
public class EnhancedTokenService {

    @Value("${token.secret}")
    private String secret;

    @Autowired
    private RedisTemplate<Object, Object> redisTemplate;

    /**
     * 创建增强版Token（包含设备指纹）
     */
    public String createEnhancedToken(Map<String, Object> claims,
                                       String deviceInfo) {
        Date now = new Date();
        // accessToken短期（30分钟）
        Date accessExpire = new Date(now.getTime() + 30 * 60 * 1000L);
        // refreshToken长期（7天）
        Date refreshExpire = new Date(now.getTime() + 7 * 24 * 60 * 60 * 1000L);

        // 生成设备指纹Hash
        String deviceFingerprint = generateDeviceFingerprint(deviceInfo);
        claims.put("deviceFingerprint", deviceFingerprint);

        // 生成accessToken
        String accessToken = Jwts.builder()
                .setClaims(claims)
                .setIssuedAt(now)
                .setExpiration(accessExpire)
                .claim("type", "access")
                .signWith(SignatureAlgorithm.HS512, secret)
                .compact();

        // 生成refreshToken
        String refreshToken = Jwts.builder()
                .setClaims(Map.of("userId", claims.get("userId")))
                .setIssuedAt(now)
                .setExpiration(refreshExpire)
                .claim("type", "refresh")
                .signWith(SignatureAlgorithm.HS512, secret)
                .compact();

        // 存储refreshToken到Redis（用于主动失效）
        String key = "token:refresh:" + claims.get("userId");
        redisTemplate.opsForValue().set(key, refreshToken, 7, TimeUnit.DAYS);

        // 返回双Token
        Map<String, Object> tokenMap = new HashMap<>();
        tokenMap.put("accessToken", accessToken);
        tokenMap.put("refreshToken", refreshToken);
        tokenMap.put("expiresIn", 1800); // 30分钟

        return JSONUtil.toJsonStr(tokenMap);
    }

    /**
     * 验证设备指纹
     */
    public boolean validateDeviceFingerprint(String token, String currentDeviceInfo) {
        Claims claims = parseTokenSafely(token);
        if (claims == null) return false;

        String storedFingerprint = claims.get("deviceFingerprint", String.class);
        String currentFingerprint = generateDeviceFingerprint(currentDeviceInfo);

        return storedFingerprint != null && storedFingerprint.equals(currentFingerprint);
    }

    /**
     * 生成设备指纹（基于User-Agent + IP前缀）
     */
    private String generateDeviceFingerprint(String deviceInfo) {
        return DigestUtil.md5Hex(deviceInfo);
    }

    /**
     * 使用refreshToken刷新accessToken
     */
    public String refreshAccessToken(String refreshToken) {
        Claims claims = parseTokenSafely(refreshToken);
        if (claims == null || !"refresh".equals(claims.get("type"))) {
            throw new BaseException("无效的refreshToken");
        }

        Long userId = claims.get("userId", Long.class);

        // 验证Redis中的refreshToken
        String key = "token:refresh:" + userId;
        String storedToken = (String) redisTemplate.opsForValue().get(key);
        if (!refreshToken.equals(storedToken)) {
            throw new BaseException("refreshToken已失效");
        }

        // 生成新的accessToken
        Map<String, Object> newClaims = new HashMap<>();
        newClaims.put("userId", userId);
        newClaims.put("deviceFingerprint", claims.get("deviceFingerprint"));

        return Jwts.builder()
                .setClaims(newClaims)
                .setIssuedAt(new Date())
                .setExpiration(new Date(System.currentTimeMillis() + 30 * 60 * 1000L))
                .claim("type", "access")
                .signWith(SignatureAlgorithm.HS512, secret)
                .compact();
    }

    /**
     * 强制下线（删除refreshToken）
     */
    public void forceLogout(Long userId) {
        String key = "token:refresh:" + userId;
        redisTemplate.delete(key);
    }

    private Claims parseTokenSafely(String token) {
        try {
            return Jwts.parser().setSigningKey(secret).parseClaimsJws(token).getBody();
        } catch (Exception e) {
            return null;
        }
    }
}

// 登录失败次数限制
@Component
public class LoginRateLimiter {

    @Autowired
    private RedisTemplate<Object, Object> redisTemplate;

    private static final int MAX_ATTEMPTS = 5;
    private static final int LOCK_DURATION_MINUTES = 15;
    private static final String PREFIX = "login:attempts:";

    /**
     * 检查是否被锁定
     */
    public void checkLocked(String identifier) {
        String key = PREFIX + identifier;
        Object count = redisTemplate.opsForValue().get(key + ":locked");
        if (count != null) {
            throw new BaseException("登录尝试次数过多，请15分钟后再试");
        }
    }

    /**
     * 记录登录失败
     */
    public void recordFailedAttempt(String identifier) {
        String key = PREFIX + identifier;
        Long attempts = redisTemplate.opsForValue().increment(key);
        redisTemplate.expire(key, 1, TimeUnit.HOURS);

        if (attempts != null && attempts >= MAX_ATTEMPTS) {
            redisTemplate.opsForValue().set(key + ":locked", "1",
                LOCK_DURATION_MINUTES, TimeUnit.MINUTES);
            log.warn("用户登录被锁定, identifier: {}", identifier);
        }
    }

    /**
     * 登录成功后清除计数
     */
    public void clearAttempts(String identifier) {
        String key = PREFIX + identifier;
        redisTemplate.delete(key);
        redisTemplate.delete(key + ":locked");
    }
}
```

```
"我在小程序登录模块用ThreadLocal存储当前用户ID。
拦截器解析Token后把userId放入ThreadLocal，
后续Controller和Service随时获取，不用层层传参。"

第二层：再说原理（20秒）
"每个Thread对象内部维护一个ThreadLocalMap，
Key是ThreadLocal对象（弱引用），Value是存储的值。
不同线程访问同一个ThreadLocal，拿到各自的值。"

第三层：重点说坑（20秒）
"最大的坑是Tomcat线程池复用。
如果请求结束后不清理ThreadLocal，
下一个请求可能拿到上一个用户的ID，
导致严重的数据安全问题。"

第四层：说解决（15秒）
"我在拦截器的afterCompletion中调用remove()，
这个方法无论成功或异常都会执行，保证100%清理。"
```
