```java
// CheckInServiceImpl.java - 入住业务实现
@Service
@Slf4j
public class CheckInServiceImpl extends ServiceImpl<CheckInMapper, CheckIn>
    implements ICheckInService {

    @Autowired
    private ElderMapper elderMapper;

    @Autowired
    private BedMapper bedMapper;

    @Autowired
    private ContractMapper contractMapper;

    @Autowired
    private CheckInConfigMapper checkInConfigMapper;

    /**
     * 办理入住（多表联动）
     *
     * 业务流程：
     * 1. 校验老人是否已入住
     * 2. 更新床位状态为已入住
     * 3. 保存或更新老人档案
     * 4. 生成合同编号并创建合同
     * 5. 写入入住记录
     * 6. 写入入住配置
     */
    @Override
    @Transactional(rollbackFor = Exception.class)
    public CheckInResultVo checkIn(CheckInDto dto) {
        log.info("开始办理入住, elderName: {}", dto.getElderName());

        // ========== 1. 前置校验 ==========

        // 1.1 校验老人是否已入住
        Elder elder = elderMapper.selectById(dto.getElderId());
        if (elder == null) {
            throw new BaseException("老人信息不存在");
        }
        if (ElderStatus.CHECKED_IN.getCode().equals(elder.getStatus())) {
            throw new BaseException("该老人已入住，请勿重复操作");
        }

        // 1.2 校验床位是否可用
        Bed bed = bedMapper.selectById(dto.getBedId());
        if (bed == null) {
            throw new BaseException("床位信息不存在");
        }
        if (!BedStatus.AVAILABLE.getCode().equals(bed.getStatus())) {
            throw new BaseException("该床位不可用，请选择其他床位");
        }

        // ========== 2. 更新床位状态 ==========
        bed.setStatus(BedStatus.OCCUPIED.getCode());
        bed.setElderId(dto.getElderId());
        bedMapper.updateById(bed);
        log.info("床位状态已更新, bedId: {}, status: OCCUPIED", bed.getId());

        // ========== 3. 更新老人档案 ==========
        elder.setStatus(ElderStatus.CHECKED_IN.getCode());
        elder.setBedId(dto.getBedId());
        elder.setBedNumber(bed.getBedNumber());
        elderMapper.updateById(elder);
        log.info("老人状态已更新, elderId: {}, status: CHECKED_IN", elder.getId());

        // ========== 4. 创建合同 ==========
        Contract contract = new Contract();
        contract.setContractNo(generateContractNo());
        contract.setElderId(dto.getElderId());
        contract.setElderName(dto.getElderName());
        contract.setContractStatus(ContractStatus.PENDING.getCode());
        contract.setStartTime(dto.getContractStartTime());
        contract.setEndTime(dto.getContractEndTime());
        contract.setTermType(dto.getTermType());
        contract.setSignatory(dto.getSignatory());
        contract.setSignatoryPhone(dto.getSignatoryPhone());
        contractMapper.insert(contract);
        log.info("合同已创建, contractNo: {}", contract.getContractNo());

        // ========== 5. 写入入住记录 ==========
        CheckIn checkIn = new CheckIn();
        checkIn.setElderId(dto.getElderId());
        checkIn.setElderName(dto.getElderName());
        checkIn.setBedId(dto.getBedId());
        checkIn.setBedNumber(bed.getBedNumber());
        checkIn.setContractId(contract.getId());
        checkIn.setCheckInStatus(CheckInStatus.ACTIVE.getCode());
        checkIn.setCheckInTime(LocalDateTime.now());
        checkIn.setRemark(dto.getRemark());
        save(checkIn);
        log.info("入住记录已创建, checkInId: {}", checkIn.getId());

        // ========== 6. 写入入住配置 ==========
        CheckInConfig config = new CheckInConfig();
        config.setCheckInId(checkIn.getId());
        config.setElderId(dto.getElderId());
        config.setBedId(dto.getBedId());
        config.setNursingLevelId(dto.getNursingLevelId());
        config.setNursingLevelName(dto.getNursingLevelName());
        config.setStartTime(dto.getContractStartTime());
        config.setEndTime(dto.getContractEndTime());
        checkInConfigMapper.insert(config);
        log.info("入住配置已创建, configId: {}", config.getId());

        // ========== 7. 返回结果 ==========
        return CheckInResultVo.builder()
            .checkInId(checkIn.getId())
            .contractNo(contract.getContractNo())
            .bedNumber(bed.getBedNumber())
            .checkInTime(checkIn.getCheckInTime())
            .build();
    }

    /**
     * 生成合同编号
     * 格式：HT + yyyyMMdd + 4位随机数
     */
    private String generateContractNo() {
        String dateStr = LocalDate.now().format(DateTimeFormatter.ofPattern("yyyyMMdd"));
        String randomStr = String.format("%04d", (int)(Math.random() * 10000));
        return "HT" + dateStr + randomStr;
    }
}
```

**详细解析**：

1. **多表联动设计要点**：
   - 6个步骤在一个事务中完成
   - 每个步骤有独立的前置校验
   - 使用`@Transactional`保证原子性

2. **状态流转规则**：
   ```
   老人状态：启用(1) → 入住中(4) → 退住中(3) → 已退住(5)
   床位状态：空闲(0) → 已入住(1) → 空闲(0)
   合同状态：待生效(0) → 生效中(1) → 已到期(2)
   ```

3. **合同编号生成**：
   - 格式：HT + 日期 + 随机数
   - 保证可读性和唯一性
   - 生产环境可加数据库唯一索引

**面试思维链**：

```
面试官问：入住办理涉及多张表，怎么保证数据一致性？

思维链：
1. 先说业务流程：6个步骤，多表联动
2. 再说技术方案：@Transactional事务保证
```

```java
            .get(CacheConstants.NURSING_LEVEL_ALL);

                .map(projectId -> {
                    NursingProjectPlan pp = new NursingProjectPlan();
                    pp.setPlanId(plan.getId());
                    pp.setProjectId(projectId);
                    return pp;
                })
                .collect(Collectors.toList());

            projectPlanMapper.insertBatch(projectPlans);
        }

        // 3. 删除缓存
        redisTemplate.delete(CacheConstants.NURSING_PLAN_ALL);
        log.info("护理计划已创建, planId: {}", plan.getId());

        }

        List<NursingProjectPlan> projectPlans = projectPlanMapper.selectList(
```

```java
// ContractStatusJob.java
@Slf4j
@Component
public class ContractStatusJob {

    @Autowired
    private IContractService contractService;

    /**
     * 每天凌晨1点执行
     */
    @Scheduled(cron = "0 0 1 * * ?")
    public void updateContractStatus() {
        log.info("========== 合同状态更新定时任务开始 ==========");

        try {
            // 1. 待生效 → 生效中
            int pendingCount = handlePendingContracts();
            log.info("处理待生效合同: {} 条", pendingCount);

            // 2. 生效中 → 已到期
            int expiredCount = handleExpiredContracts();
            log.info("处理已到期合同: {} 条", expiredCount);

        } catch (Exception e) {
            log.error("合同状态更新定时任务执行失败", e);
        }

        log.info("========== 合同状态更新定时任务结束 ==========");
    }

    /**
     * 处理待生效合同
     */
    private int handlePendingContracts() {
        LocalDateTime now = LocalDateTime.now();

        List<Contract> pendingContracts = contractService.list(
            Wrappers.<Contract>lambdaQuery()
                .eq(Contract::getContractStatus, ContractStatus.PENDING.getCode())
                .le(Contract::getStartTime, now)
        );

        if (CollUtil.isEmpty(pendingContracts)) {
            return 0;
        }

        for (Contract contract : pendingContracts) {
            contract.setContractStatus(ContractStatus.ACTIVE.getCode());
            contractService.updateById(contract);
        }

        return pendingContracts.size();
    }

    /**
     * 处理已到期合同
     */
    private int handleExpiredContracts() {
        LocalDateTime now = LocalDateTime.now();

        List<Contract> expiredContracts = contractService.list(
            Wrappers.<Contract>lambdaQuery()
                .eq(Contract::getContractStatus, ContractStatus.ACTIVE.getCode())
                .le(Contract::getEndTime, now)
        );

        if (CollUtil.isEmpty(expiredContracts)) {
            return 0;
        }

        for (Contract contract : expiredContracts) {
            contract.setContractStatus(ContractStatus.EXPIRED.getCode());
            contractService.updateById(contract);
        }

        return expiredContracts.size();
    }
}
```

```java
    @Override
    public PageResult<NursingProjectVo> pageQuery(int pageNum, int pageSize, String name) {
        // 1. 构建查询条件
        LambdaQueryWrapper<NursingProject> wrapper = new LambdaQueryWrapper<>();
        wrapper.eq(NursingProject::getStatus, 1); // 启用状态
        if (StringUtils.isNotEmpty(name)) {
            wrapper.like(NursingProject::getName, name);
        }
        wrapper.orderByAsc(NursingProject::getSort);

        // 2. 分页查询
        Page<NursingProject> page = new Page<>(pageNum, pageSize);
        Page<NursingProject> resultPage = page(page, wrapper);

        // 3. 转换VO
        List<NursingProjectVo> vos = BeanUtil.copyToList(
            resultPage.getRecords(), NursingProjectVo.class);

        return new PageResult<>(vos, resultPage.getTotal());
    }

        NursingProject project = getById(id);
        if (project == null) {
            throw new BaseException("护理项目不存在");
        }

        NursingProjectDetailVo vo = BeanUtil.toBean(project, NursingProjectDetailVo.class);

        // 查询关联的护理计划
        List<NursingProjectPlan> projectPlans = projectPlanMapper.selectList(
            Wrappers.<NursingProjectPlan>lambdaQuery()
                .eq(NursingProjectPlan::getProjectId, id)

                .map(NursingProjectPlan::getPlanId)
                .collect(Collectors.toList());
            List<NursingPlan> plans = nursingPlanMapper.selectBatchIds(planIds);
            vo.setPlans(BeanUtil.copyToList(plans, NursingPlanSimpleVo.class));
        }
```

    }

// ReservationServiceImpl.java
@Service
@Slf4j
public class ReservationServiceImpl
    extends ServiceImpl<ReservationMapper, Reservation>
    implements IReservationService {

    private static final int MAX_CANCEL_PER_DAY = 3;  // 每天最多取消3次
    private static final int MAX_PER_SLOT = 10;       // 每个时间段最多10人

    @Override
    public int getCancelledCount(Long userId) {
        LocalDate today = LocalDate.now();
        return (int) count(
            Wrappers.<Reservation>lambdaQuery()
                .eq(Reservation::getUserId, userId)
                .eq(Reservation::getReservationDate, today)
                .eq(Reservation::getStatus, ReservationStatus.CANCELLED.getCode())
        );
    }

    @Override
    public List<TimeSlotVo> countByTime() {
        LocalDate today = LocalDate.now();
        List<TimeSlotVo> result = new ArrayList<>();

        // 上午时间段
        int morningCount = (int) count(
            Wrappers.<Reservation>lambdaQuery()
                .eq(Reservation::getReservationDate, today)
                .eq(Reservation::getTimeSlot, 0)
                .ne(Reservation::getStatus, ReservationStatus.CANCELLED.getCode())
        );
        result.add(new TimeSlotVo("上午", MAX_PER_SLOT - morningCount));

        // 下午时间段
        int afternoonCount = (int) count(
            Wrappers.<Reservation>lambdaQuery()
                .eq(Reservation::getReservationDate, today)
                .eq(Reservation::getTimeSlot, 1)
                .ne(Reservation::getStatus, ReservationStatus.CANCELLED.getCode())
        );
        result.add(new TimeSlotVo("下午", MAX_PER_SLOT - afternoonCount));

        return result;
    }

    @Override
    @Transactional(rollbackFor = Exception.class)
    public void createReservation(ReservationDto dto, Long userId) {
        // 1. 校验取消次数
        int cancelledCount = getCancelledCount(userId);
        if (cancelledCount >= MAX_CANCEL_PER_DAY) {
            throw new BaseException("今日取消预约已达" + MAX_CANCEL_PER_DAY + "次，无法继续预约");
        }

        // 2. 校验时间段剩余次数
        LocalDate today = LocalDate.now();
        int slotCount = (int) count(
            Wrappers.<Reservation>lambdaQuery()
                .eq(Reservation::getReservationDate, today)
                .eq(Reservation::getTimeSlot, dto.getTimeSlot())
                .ne(Reservation::getStatus, ReservationStatus.CANCELLED.getCode())
        );
        if (slotCount >= MAX_PER_SLOT) {
            throw new BaseException("该时间段预约已满");
        }

        // 3. 创建预约
        Reservation reservation = new Reservation();
        reservation.setUserId(userId);
        reservation.setUserName(dto.getUserName());
        reservation.setUserPhone(dto.getUserPhone());
        reservation.setReservationDate(today);
        reservation.setTimeSlot(dto.getTimeSlot());
        reservation.setStatus(ReservationStatus.BOOKED.getCode());
        reservation.setRemark(dto.getRemark());
        save(reservation);

        log.info("预约创建成功, id: {}, userId: {}", reservation.getId(), userId);
    }

    @Override
    public PageResult<ReservationVo> pageQuery(int pageNum, int pageSize, Long userId) {
        Page<Reservation> page = new Page<>(pageNum, pageSize);

        Page<Reservation> resultPage = page(page,
            Wrappers.<Reservation>lambdaQuery()
                .eq(Reservation::getUserId, userId)
                .orderByDesc(Reservation::getCreateTime)
        );

        List<ReservationVo> vos = BeanUtil.copyToList(
            resultPage.getRecords(), ReservationVo.class);

        return new PageResult<>(vos, resultPage.getTotal());
    }

    @Override
    @Transactional(rollbackFor = Exception.class)
    public void cancel(Long id, Long userId) {
        Reservation reservation = getById(id);
        if (reservation == null) {
            throw new BaseException("预约不存在");
        }
        if (!reservation.getUserId().equals(userId)) {
            throw new BaseException("只能取消自己的预约");
        }
        if (!ReservationStatus.BOOKED.getCode().equals(reservation.getStatus())) {
            throw new BaseException("只能取消已预约状态的预约");
        }

        reservation.setStatus(ReservationStatus.CANCELLED.getCode());
        updateById(reservation);

        log.info("预约已取消, id: {}, userId: {}", id, userId);
    }
}
