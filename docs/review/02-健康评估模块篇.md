```java
@ApiOperation("上传体检报告")
@PostMapping("/upload")
public AjaxResult uploadFile(@RequestPart("file") MultipartFile file,
                             @RequestPart(value = "idCard") String idCardNo) {
    try {
        // 1. 文件校验
        validateFile(file);

        // 2. 上传到阿里云OSS
        String url = aliyunOSSOperator.upload(file.getBytes(), file.getOriginalFilename());

        // 3. PDF文件内容读取为字符串
        String content = PDFUtil.pdfToString(file.getInputStream());

        // 4. 临时存储到Redis中（Hash结构）
        redisTemplate.opsForHash().put("healthReport", idCardNo, content);

        // 5. 设置24小时过期时间
        redisTemplate.expire("healthReport", 24, TimeUnit.HOURS);

        return AjaxResult.success().put("url", url);
    } catch (Exception e) {
        log.error("上传体检报告失败, idCard: {}", idCardNo, e);
        return AjaxResult.error(e.getMessage());
    }
}

/**
 * 文件校验
 */
private void validateFile(MultipartFile file) {
    if (file.isEmpty()) {
        throw new BaseException("上传文件不能为空");
    }

    // 校验文件类型
    String originalFilename = file.getOriginalFilename();
    if (originalFilename == null || !originalFilename.toLowerCase().endsWith(".pdf")) {
        throw new BaseException("只支持PDF格式文件");
    }

    // 校验文件大小（限制10MB）
    if (file.getSize() > 10 * 1024 * 1024) {
        throw new BaseException("文件大小不能超过10MB");
    }
}
```

```java
// PDFUtil.java - PDF解析工具类
public class PDFUtil {
    public static String pdfToString(InputStream inputStream) {
        PDDocument document = null;
        try {
            // 加载PDF文档
            document = PDDocument.load(inputStream);

            // 创建PDFTextStripper提取文本
            PDFTextStripper pdfStripper = new PDFTextStripper();
            // 设置排序（按页面顺序）
            pdfStripper.setSortByPosition(true);
            // 设置起始页和结束页
            // pdfStripper.setStartPage(1);
            // pdfStripper.setEndPage(document.getNumberOfPages());

            // 提取文本内容
            String text = pdfStripper.getText(document);

            // 基本清洗：去除多余空白
            text = text.replaceAll("\\n{3,}", "\n\n");  // 多个换行合并
            text = text.replaceAll(" {2,}", " ");        // 多个空格合并

            return text;
        } catch (IOException e) {
            log.error("PDF解析失败", e);
            throw new BaseException("PDF文件解析失败，请检查文件是否损坏");
        } finally {
            // 资源释放
            if (document != null) {
                try {
                    document.close();
                } catch (IOException e) {
                    log.error("关闭PDF文档失败", e);
                }
            }
            try {
                inputStream.close();
            } catch (IOException e) {
                log.error("关闭输入流失败", e);
            }
        }
    }
}
```

```java
// AliyunOSSOperator.java - OSS操作封装
@Component
public class AliyunOSSOperator {

    @Autowired
    private OSSClient ossClient;

    @Value("${aliyun.oss.bucketName}")
    private String bucketName;

    @Value("${aliyun.oss.endpoint}")
    private String endpoint;

    /**
     * 上传文件到OSS
     */
    public String upload(byte[] fileBytes, String originalFilename) {
        // 生成唯一文件名
        String fileName = generateFileName(originalFilename);

        try {
            // 上传到OSS
            ossClient.putObject(bucketName, fileName, new ByteArrayInputStream(fileBytes));

            // 返回访问URL
            return String.format("https://%s.%s/%s", bucketName, endpoint, fileName);
        } catch (Exception e) {
            log.error("上传文件到OSS失败", e);
            throw new BaseException("文件上传失败，请稍后重试");
        }
    }

    /**
     * 生成唯一文件名
     */
    private String generateFileName(String originalFilename) {
        String extension = originalFilename.substring(originalFilename.lastIndexOf("."));
        String datePath = LocalDate.now().format(DateTimeFormatter.ofPattern("yyyy/MM/dd"));
        String uuid = UUID.randomUUID().toString().replace("-", "");
        return String.format("health-report/%s/%s%s", datePath, uuid, extension);
    }
}
```

**详细解析**：

1. **PDF解析技术选型**：
   - **PDFBox**：Apache开源，纯Java实现，适合文本提取
   - **iText**：功能强大但商业版收费
   - **PDF.js**：前端解析，不适合后端批量处理

2. **Redis缓存设计考量**：
   - **为什么用Hash**：一个Key管理所有用户报告，Field按身份证号快速定位
   - **为什么24小时过期**：用户通常当天完成评估，避免长期占用内存
   - **为什么不上传即评估**：上传和评估是两个交互动作，分离更灵活

3. **文件上传安全**：
   - 文件类型校验：只允许PDF
   - 文件大小限制：防止恶意大文件攻击
   - 文件名处理：使用UUID重命名，防止路径遍历

4. **OSS存储设计**：
   - 按日期分目录：`health-report/2026/06/16/uuid.pdf`
   - 便于管理、归档和CDN加速

**面试思维链**：

```
面试官问：PDF体检报告是怎么处理的？

思维链：
1. 先说整体流程：上传 → OSS存储 → PDF解析 → Redis缓存
2. 再说技术细节：PDFBox解析、Hash结构缓存、24小时过期
3. 最后说为什么这样设计

回答要点：
"用户上传PDF后，系统做了三件事：
1. 文件上传到阿里云OSS，按日期分目录存储
2. 使用Apache PDFBox解析PDF内容为纯文本
3. 文本内容按身份证号存入Redis Hash，24小时过期

这样设计的原因是：
- 上传和评估是两个步骤，分离后用户可以先上传再慢慢填写表单
- Redis缓存避免重复解析PDF（每次解析2-3秒）
- 24小时过期平衡了用户体验和内存资源
"

A: 只能提取文本层，扫描件PDF无法处理，需要OCR技术补充

Q: 如果Redis缓存过期了怎么办？
A: 提示用户重新上传，同时记录日志排查Redis问题
```

```java
// HealthAssessmentServiceImpl.java - 核心评估逻辑
@Service
public class HealthAssessmentServiceImpl extends ServiceImpl<HealthAssessmentMapper, HealthAssessment>
    implements IHealthAssessmentService {

    @Autowired
    private RedisTemplate<Object, Object> redisTemplate;

    @Autowired
    private AIModelInvoker aiModelInvoker;

    @Override
    public Long insertHealthAssessment(HealthAssessment healthAssessment) {
        // 1. 从Redis获取体检报告内容，构建Prompt
        String prompt = getPrompt(healthAssessment.getIdCard());

        // 2. 调用百度千帆AI大模型
        String qianfanResult = aiModelInvoker.qianfanInvoker(prompt);

        // 3. 解析AI返回的JSON结果
        HealthReportVo healthReportVo;
        try {
            healthReportVo = JSON.parseObject(qianfanResult, HealthReportVo.class);
        } catch (Exception e) {
            log.error("AI返回结果解析失败, result: {}", qianfanResult, e);
            throw new BaseException("AI分析结果解析失败，请稍后重试");
        }

        // 4. 保存评估结果到数据库
        return saveHealthAssessment(healthReportVo, healthAssessment);
    }

    /**
     * 构建Prompt提示词
     */
    private String getPrompt(String idCard) {
        // 从Redis获取体检报告内容
        String content = (String) redisTemplate.opsForHash().get("healthReport", idCard);

        if (StringUtils.isEmpty(content)) {
            log.error("从Redis中未获取到身份证号[{}]的体检报告内容", idCard);
            throw new BaseException("未找到体检报告，请先上传PDF报告后再提交");
        }

        // 构建强约束Prompt
        return buildHealthAssessmentPrompt(content);
    }

    /**
     * 构建健康评估Prompt
     * 设计原则：
     * 1. 角色设定：专业医生视角
     * 2. 任务分解：分步骤明确要求
     * 3. 输出约束：强制JSON格式
     * 4. 字段固定：每个字段都有明确说明

    private String buildHealthAssessmentPrompt(String reportContent) {
        StringBuilder prompt = new StringBuilder();
        prompt.append("你是一位经验丰富的体检中心主任医师，请以专业医生的视角分析以下体检报告。\n\n");

        prompt.append(reportContent).append("\n\n");

        prompt.append("请严格按照以下要求进行分析，并只返回JSON格式结果，不要包含任何其他文字：\n\n");

        prompt.append("1. 提取体检报告中的\"总检日期\"；\n\n");

        prompt.append("2. 给出健康评估结果：\n");
        prompt.append("   - riskLevel: 风险等级，必须是以下之一：\"健康\"、\"提示\"、\"风险\"、\"危险\"、\"严重危险\"\n");
        prompt.append("   - healthIndex: 健康指数，0-100之间的数字，分数越高越健康\n\n");

        prompt.append("3. 风险等级占比分析（riskDistribution）：\n");
        prompt.append("   - healthy: 健康占比\n");
        prompt.append("   - caution: 提示占比\n");
        prompt.append("   - risk: 风险占比\n");
        prompt.append("   - danger: 危险占比\n");
        prompt.append("   - severeDanger: 严重危险占比\n");
        prompt.append("   五项之和必须等于100\n\n");

        prompt.append("4. 异常数据详细解读（abnormalData数组）：\n");
        prompt.append("   每一项包含：conclusion(结论), examinationItem(检查项目), result(结果),\n");
```

```java
// IDCardUtils.java - 身份证号解析工具
public class IDCardUtils {

    // 身份证号长度
    private static final int ID_CARD_LENGTH_15 = 15;
    private static final int ID_CARD_LENGTH_18 = 18;

    // 加权因子（18位身份证）
    private static final int[] WEIGHT_FACTORS = {7, 9, 10, 5, 8, 4, 2, 1, 6, 3, 7, 9, 10, 5, 8, 4, 2};

    // 校验码
    private static final char[] CHECK_CODES = {'1', '0', 'X', '9', '8', '7', '6', '5', '4', '3', '2'};

    /**
     * 从身份证号提取出生日期
     */
    public static LocalDateTime getBirthDate(String idCard) {
        if (StringUtils.isEmpty(idCard)) {
            return null;
        }

        String birthDateStr;
        if (idCard.length() == ID_CARD_LENGTH_18) {
            // 18位：第7-14位是出生日期
            birthDateStr = idCard.substring(6, 14);
        } else if (idCard.length() == ID_CARD_LENGTH_15) {
            // 15位：第7-12位是出生日期（年份只有2位）
            birthDateStr = "19" + idCard.substring(6, 12);
        } else {
            throw new BaseException("身份证号格式不正确");
        }

        try {
            DateTimeFormatter formatter = DateTimeFormatter.ofPattern("yyyyMMdd");
            LocalDate birthDate = LocalDate.parse(birthDateStr, formatter);
            return birthDate.atStartOfDay();
        } catch (Exception e) {
            throw new BaseException("身份证号中出生日期格式不正确");
        }
    }

    /**
     * 从身份证号计算年龄
     */
    public static Integer getAge(String idCard) {
        LocalDateTime birthDate = getBirthDate(idCard);
        if (birthDate == null) {
            return null;
        }

        LocalDate birthLocalDate = birthDate.toLocalDate();
        LocalDate now = LocalDate.now();

        // 计算周岁
        int age = now.getYear() - birthLocalDate.getYear();
        // 如果今年生日还没过，年龄减1
        if (birthLocalDate.plusYears(age).isAfter(now)) {
            age--;
        }

        return age;
    }

    /**
     * 从身份证号提取性别
     * @return 0:男, 1:女
     */
    public static Integer getGender(String idCard) {
        if (StringUtils.isEmpty(idCard)) {
            return null;
        }

        char genderChar;
        if (idCard.length() == ID_CARD_LENGTH_18) {
            // 18位：第17位是性别位
            genderChar = idCard.charAt(16);
        } else if (idCard.length() == ID_CARD_LENGTH_15) {
            // 15位：第15位是性别位
            genderChar = idCard.charAt(14);
        } else {
            throw new BaseException("身份证号格式不正确");
        }

        // 奇数为男，偶数为女
        int genderNum = Character.getNumericValue(genderChar);
        return genderNum % 2 == 1 ? 0 : 1;  // 0:男, 1:女
    }

    /**
     * 验证身份证号格式
     */
    public static boolean isValid(String idCard) {
        if (StringUtils.isEmpty(idCard)) {
            return false;
        }

        if (idCard.length() == ID_CARD_LENGTH_18) {
            return validate18IdCard(idCard);
        } else if (idCard.length() == ID_CARD_LENGTH_15) {
            return validate15IdCard(idCard);
        }

        return false;
    }

    /**
     * 验证18位身份证号
     */
    private static boolean validate18IdCard(String idCard) {
        // 前17位必须是数字
        String idCard17 = idCard.substring(0, 17);
        if (!idCard17.matches("\\d{17}")) {
            return false;
        }

        // 计算校验码
        int sum = 0;
        for (int i = 0; i < 17; i++) {
            sum += Character.getNumericValue(idCard.charAt(i)) * WEIGHT_FACTORS[i];
        }
        int mod = sum % 11;
        char checkCode = CHECK_CODES[mod];

        // 比较校验码
        return idCard.charAt(17) == checkCode;
    }

    /**
     * 验证15位身份证号
     */
    private static boolean validate15IdCard(String idCard) {
        return idCard.matches("\\d{15}");
    }

    /**
     * 脱敏处理（用于日志输出）
     */
    public static String mask(String idCard) {
        if (StringUtils.isEmpty(idCard) || idCard.length() < 8) {
            return idCard;
        }
        // 保留前4位和后4位，中间用*代替
        return idCard.substring(0, 4) + "**********" + idCard.substring(idCard.length() - 4);
    }
```

```java
    // 60分以下不建议入住，60分以上建议入住
    healthAssessment.setSuggestionForAdmission(healthIndex >= 60 ? 0 : 1);

    // 根据实际入住状态设置
    healthAssessment.setAdmissionStatus(1); // 默认未入住

    // 7. 设置AI分析结果
    healthAssessment.setHealthScore(String.valueOf(healthIndex));
    healthAssessment.setRiskLevel(healthReportVo.getHealthAssessment().getRiskLevel());
    healthAssessment.setTotalCheckDate(healthReportVo.getTotalCheckDate());
    healthAssessment.setAssessmentTime(LocalDateTime.now());
    healthAssessment.setReportSummary(healthReportVo.getSummarize());

    healthAssessment.setAbnormalAnalysis(JSON.toJSONString(healthReportVo.getAbnormalData()));
    healthAssessment.setSystemScore(JSON.toJSONString(healthReportVo.getSystemScore()));

    save(healthAssessment);
```

**详细解析**：

1. **身份证号编码规则**：
   - 18位：前6位地址码 + 8位出生日期 + 3位顺序码 + 1位校验码
   - 15位：前6位地址码 + 6位出生日期 + 3位顺序码
   - 性别：第17位（18位）/ 第15位（15位），奇数为男，偶数为女

2. **业务规则与AI结果结合的设计理念**：
   - **AI负责**：认知型分析（风险等级、异常指标、系统评分）
   - **后端负责**：确定性规则（身份证解析、护理等级映射、入住建议）
   - **分离原因**：业务规则需要确定性，AI输出有概率性

3. **工具类设计原则**：
   - 静态方法，无需实例化
   - 空值安全，返回null而非抛异常
   - 脱敏处理，日志输出时保护隐私

```
- AI负责：分析体检报告，输出健康分数、风险等级、异常指标等
```

```java
// 方案一：Spring @Async 异步处理（简单场景）
@Service
public class HealthAssessmentAsyncService {

    @Autowired
    private AIModelInvoker aiModelInvoker;

    @Autowired
    private HealthAssessmentMapper healthAssessmentMapper;

    @Autowired
    private RedisTemplate<Object, Object> redisTemplate;

    /**
     * 异步执行健康评估
     */
    @Async("healthAssessmentExecutor")
    public CompletableFuture<Long> asyncAssess(String idCard, String elderName) {
        try {
            // 1. 从Redis获取报告内容
            String content = (String) redisTemplate.opsForHash().get("healthReport", idCard);
            if (StringUtils.isEmpty(content)) {
                updateAssessmentStatus(idCard, "FAILED", "报告缓存已过期");
                return CompletableFuture.completedFuture(null);
            }

            // 2. 构建Prompt并调用AI
            String prompt = buildPrompt(content);
            String aiResult = aiModelInvoker.qianfanInvoker(prompt);

            // 3. 解析结果
            HealthReportVo vo = JSON.parseObject(aiResult, HealthReportVo.class);

            // 4. 保存结果
            HealthAssessment assessment = buildAssessment(vo, idCard, elderName);
            healthAssessmentMapper.insert(assessment);

            // 5. 更新状态为完成
            updateAssessmentStatus(idCard, "COMPLETED", null);

            return CompletableFuture.completedFuture(assessment.getId());

        } catch (Exception e) {
            log.error("异步健康评估失败, idCard: {}", idCard, e);
            updateAssessmentStatus(idCard, "FAILED", e.getMessage());
            return CompletableFuture.completedFuture(null);
        }
    }

    /**
     * 更新评估状态（存储到Redis）
     */
    private void updateAssessmentStatus(String idCard, String status, String errorMsg) {
        Map<String, String> statusMap = new HashMap<>();
        statusMap.put("status", status);
        statusMap.put("errorMsg", errorMsg != null ? errorMsg : "");
        statusMap.put("updateTime", LocalDateTime.now().toString());

        redisTemplate.opsForHash().putAll("health:assessment:status:" + idCard, statusMap);
        redisTemplate.expire("health:assessment:status:" + idCard, 1, TimeUnit.HOURS);
    }
}

// 异步线程池配置
@Configuration
@EnableAsync
public class AsyncConfig {

    @Bean("healthAssessmentExecutor")
    public Executor healthAssessmentExecutor() {
        ThreadPoolTaskExecutor executor = new ThreadPoolTaskExecutor();
        executor.setCorePoolSize(2);        // 核心线程数
        executor.setMaxPoolSize(5);         // 最大线程数
        executor.setQueueCapacity(100);     // 队列容量
        executor.setThreadNamePrefix("health-assessment-");
        executor.setRejectedExecutionHandler(new ThreadPoolExecutor.CallerRunsPolicy());
        executor.initialize();
        return executor;
    }
}

@PostMapping("/submit/async")
public AjaxResult submitAsync(@RequestBody HealthAssessment assessment) {
    // 1. 参数校验
    if (StringUtils.isEmpty(assessment.getIdCard())) {
        return AjaxResult.error("身份证号不能为空");
    }

    // 2. 检查是否已有进行中的评估
    String statusKey = "health:assessment:status:" + assessment.getIdCard();
    Object status = redisTemplate.opsForHash().get(statusKey, "status");
    if ("PROCESSING".equals(status)) {
        return AjaxResult.error("该用户的健康评估正在进行中，请稍后查看结果");
    }

    // 3. 设置初始状态
    Map<String, String> statusMap = new HashMap<>();
    statusMap.put("status", "PROCESSING");
    statusMap.put("createTime", LocalDateTime.now().toString());
    redisTemplate.opsForHash().putAll(statusKey, statusMap);
    redisTemplate.expire(statusKey, 1, TimeUnit.HOURS);

    // 4. 异步执行评估
    healthAssessmentAsyncService.asyncAssess(assessment.getIdCard(), assessment.getElderName());

    return AjaxResult.success("评估已提交，请稍后查看结果");
}
```

```java
@Data
public class AssessmentMessage {
    private String idCard;       // 身份证号
    private String elderName;    // 老人姓名
    private String messageId;    // 消息ID（幂等性）
    private LocalDateTime createTime; // 创建时间

// 消息生产者
@Component
public class AssessmentProducer {

    @Autowired
    private RabbitTemplate rabbitTemplate;

    public void sendAssessmentTask(String idCard, String elderName) {
        AssessmentMessage message = new AssessmentMessage();
        message.setIdCard(idCard);
        message.setElderName(elderName);
        message.setMessageId(UUID.randomUUID().toString());
        message.setCreateTime(LocalDateTime.now());

        rabbitTemplate.convertAndSend(
            "health.assessment.exchange",
            "health.assessment.routing",
            message
        );

        log.info("发送健康评估任务, messageId: {}", message.getMessageId());
    }
}

// 消息消费者
@Component
@Slf4j
public class AssessmentConsumer {

    @Autowired
    private AIModelInvoker aiModelInvoker;

    @Autowired
    private HealthAssessmentMapper healthAssessmentMapper;

    @Autowired
    private RedisTemplate<Object, Object> redisTemplate;

    @RabbitListener(queues = "health.assessment.queue")
    public void handleAssessmentTask(AssessmentMessage message) {
        // 幂等性检查
        String idempotentKey = "health:assessment:processed:" + message.getMessageId();
        Boolean exists = redisTemplate.hasKey(idempotentKey);
        if (Boolean.TRUE.equals(exists)) {
            log.info("消息已处理，跳过, messageId: {}", message.getMessageId());
            return;
        }

        try {
            log.info("开始处理健康评估任务, messageId: {}", message.getMessageId());

            // 1. 获取报告内容
            String content = (String) redisTemplate.opsForHash()
                .get("healthReport", message.getIdCard());

            // 2. 调用AI分析
            String prompt = buildPrompt(content);
            String aiResult = aiModelInvoker.qianfanInvoker(prompt);

            // 3. 保存结果
            HealthReportVo vo = JSON.parseObject(aiResult, HealthReportVo.class);
            HealthAssessment assessment = buildAssessment(vo, message.getIdCard(), message.getElderName());
            healthAssessmentMapper.insert(assessment);

            // 4. 标记已处理（幂等性）
            redisTemplate.opsForValue().set(idempotentKey, "1", 24, TimeUnit.HOURS);

            // 5. 更新状态
            updateStatus(message.getIdCard(), "COMPLETED");

            log.info("健康评估任务处理完成, messageId: {}", message.getMessageId());

        } catch (Exception e) {
            log.error("健康评估任务处理失败, messageId: {}", message.getMessageId(), e);
            updateStatus(message.getIdCard(), "FAILED");

            // 重试机制：根据重试次数决定是否放入死信队列
            // throw new AmqpRejectAndDontRequeueException("处理失败，不重新入队");
        }
    }
}
```
